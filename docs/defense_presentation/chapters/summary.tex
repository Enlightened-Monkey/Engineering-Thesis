\section{Podsumowanie}

\begin{frame}{Główne osiągnięcia pracy}
\textbf{Realizacja celów:}

\begin{enumerate}
    \item \textbf{Analiza teoretyczna}
    \begin{itemize}
        \item Twierdzenie o redukcji polityk do postaci $(\mu, \phi_s)$
        \item Dowód kontrakcji operatora Bellmana dla dyskontowania QH
        \item Istnienie deterministycznych optymalnych polityk
    \end{itemize}
    
    \vspace{0.3em}
    \item \textbf{Algorytmy uczenia}
    \begin{itemize}
        \item \textbf{QH Policy Evaluation} -- ocena polityki dla dyskontowania QH
        \item \textbf{QH Q-Learning} -- uczenie optymalnej polityki bez modelu
        \item Zbieżność prawie na pewno (warunki Robbinsa-Monro)
    \end{itemize}
    
    \vspace{0.3em}
    \item \textbf{Weryfikacja eksperymentalna}
    \begin{itemize}
        \item MDP z dwoma stanami -- weryfikacja analityczna
        \item Zarządzanie zapasami -- test w realistycznym scenariuszu
        \item Problem rozbitka -- test skalowalności (11~616 stanów)
    \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Kluczowe wnioski}
\textbf{Niespójność czasowa w optymalnych politykach:}

\begin{block}{Obserwacja}
Dla $\alpha < 1$ optymalna reguła pierwszego kroku $\mu^\star$ różni się od optymalnej polityki stacjonarnej $\phi_s^\star$.
\end{block}

\vspace{0.3em}
\textbf{Implikacje praktyczne:}
\begin{itemize}
    \item Decydent zobowiązany (precommitted) osiąga lepsze wyniki niż naiwny
    \item Planowanie z wyprzedzeniem i trzymanie się planu jest korzystne
    \item Ważne dla:
    \begin{itemize}
        \item Ekonomii behawioralnej (oszczędności, inwestycje)
        \item Psychologii (terapia uzależnień, zmiana nawyków)
        \item Projektowania systemów wspomagania decyzji (commitment devices)
    \end{itemize}
\end{itemize}

\vspace{0.3em}
\textbf{Przykład z życia:} Student uczący się do egzaminu -- decydent zobowiązany tworzy plan nauki i go realizuje, osiągając lepszy wynik niż student codziennie odkładający naukę „na jutro".
\end{frame}

\begin{frame}{Porównanie typów decydentów}
\begin{table}[h!]
\centering
\small
\begin{tabular}{lp{5cm}p{4cm}}
\toprule
\textbf{Typ} & \textbf{Charakterystyka} & \textbf{Polityka} \\
\midrule
\textbf{Standardowy} & $\alpha = 1$, czasowo spójny & $\mu = \phi_s$ \\
\midrule
\textbf{Naiwny} & $0 < \alpha < 1$, nie przewiduje zmian preferencji & Zawsze wybiera $\mu^\star$ \\
\midrule
\textbf{Zobowiązany} & $0 < \alpha < 1$, trzyma się wcześniejszego planu & $(\mu^\star, \phi_s^\star)$ \\
\midrule
\textbf{Wyrafinowany} & $0 < \alpha < 1$, przewiduje przyszłe zmiany & Strategia uwzględniająca przyszłe ja \\
\bottomrule
\end{tabular}
\end{table}

\vspace{0.3em}
\textbf{W pracy:} Skupiliśmy się na decydentach zobowiązanych (precommitted agents), którzy mogą ustalić plan działania i go realizować.
\end{frame}

\vspace{0.5em}
\textbf{Kluczowe obserwacje:}
\begin{itemize}
    \item Algorytmy zbiegają do wartości teoretycznych (MDP 2-stan., zarządzanie zapasami)
    \item W problemie rozbitka osiągamy przeżywalność 43\% (vs. 12\% losowa, 42\% człowiek)
    \item Niespójność czasowa widoczna we wszystkich środowiskach ($\mu^\star \neq \phi_s^\star$)
    \item Dwuskalowa aproksymacja stochastyczna działa efektywnie
\end{itemize}
\end{frame}

\begin{frame}{Dalsze kierunki badań}
\textbf{Rozszerzenia teoretyczne:}
\begin{itemize}
    \item Analiza decydentów wyrafinowanych (sophisticated agents)
    \item Gry wieloagentowe z dyskontowaniem QH
    \item Nieskończone przestrzenie stanów i akcji
    \item POMDP z dyskontowaniem quasi-hiperbolicznym
\end{itemize}

\vspace{0.3em}
\textbf{Zaawansowane algorytmy:}
\begin{itemize}
    \item Deep Q-Learning dla dużych przestrzeni stanów
    \item Actor-Critic dla ciągłych przestrzeni akcji
    \item Transfer learning między różnymi wartościami $\alpha$
    \item Online adaptation parametrów $\alpha$ i $\beta$
\end{itemize}

\vspace{0.3em}
\textbf{Zastosowania praktyczne:}
\begin{itemize}
    \item Systemy rekomendacji z uwzględnieniem preferencji czasowych
    \item Planowanie finansowe osobiste (oszczędności, kredyty, emerytury)
    \item Optymalizacja łańcucha dostaw z niespójnością czasową
    \item Commitment devices w aplikacjach zdrowotnych i edukacyjnych
\end{itemize}
\end{frame}

\begin{frame}{Wyzwania i ograniczenia}
\textbf{Wyzwania badawcze:}
\begin{itemize}
    \item \textbf{Estymacja parametrów:} Jak wyznaczać $\alpha$ i $\beta$ z danych rzeczywistych?
    \item \textbf{Skalowalność:} Duże przestrzenie stanów wymagają aproksymacji funkcyjnej
    \item \textbf{Adaptacja online:} Czy parametry QH mogą się zmieniać w czasie?
    \item \textbf{Weryfikacja:} Trudność w testowaniu na rzeczywistych użytkownikach
\end{itemize}

\vspace{0.5em}
\textbf{Ograniczenia obecnego podejścia:}
\begin{itemize}
    \item Skupienie na decydentach zobowiązanych (precommitted)
    \item Skończone przestrzenie stanów i akcji
    \item Brak uwzględnienia niepewności w modelach środowiska
    \item Długi czas treningu dla dużych przestrzeni stanów
\end{itemize}

\vspace{0.5em}
\textbf{Możliwości interdyscyplinarne:}
\begin{itemize}
    \item Współpraca z ekonomistami behawioralnymi
    \item Badania kliniczne w psychologii uzależnień
    \item Projektowanie eksperymentów z rzeczywistymi uczestnikami
\end{itemize}
\end{frame}
