\section{Wprowadzenie}

\begin{frame}{Motywacja i cel pracy}
\textbf{Motywacja:}   
\begin{itemize}
    \item \textbf{Uczenie maszynowe poprzez wzmocnienie (RL)}
    \vspace{0.3em}
    \item \textbf{Połączenie psychologii z matematyką}
    \begin{itemize}
        \item Ekonomia behawioralna
        \item Zjawisko prokrastynacji
    \end{itemize}
    \vspace{0.3em}
    \item \textbf{Dyskontowanie quasi-hiperboliczne (QH):} $d(0) = 1$, $d(t) = \alpha\beta^t$ dla $t \geq 1$
    \begin{itemize}
        \item Modeluje uprzedzenie teraźniejszości (present bias)
        \item Umożliwia analizę niespójności czasowej
        \item Zastosowania: ekonomia behawioralna, psychologia, zarządzanie zasobami
    \end{itemize}
    \vspace{0.3em}
    \item \textbf{Cel pracy:} Analiza i implementacja algorytmów RL dla MDP z dyskontowaniem QH
\end{itemize}
\end{frame}

\begin{frame}{Markowski proces decyzyjny (MDP)}
\textbf{Model:} $M = (S, A, q, r, \alpha, \beta)$
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{itemize}
    \item $S$ -- skończony zbiór stanów
    \item $A$ -- skończony zbiór akcji
    \item $q(s' \mid s, a)$ -- prawdopodobieństwo przejścia
    \item $r(s, a)$ -- funkcja nagrody
    \item $\alpha > 0$ -- parametr uprzedzenia teraźniejszości
    \item $\beta \in [0,1)$ -- współczynnik dyskontowania
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\centering
\includegraphics[width=0.95\textwidth]{../thesis/fig/schematRL.png}
\small Schemat interakcji agent--środowisko
\end{column}
\end{columns}

\vspace{0.5em}
\textbf{Funkcja dyskontująca:}
\[
d(t) = 
\begin{cases}
    1, & \text{jeśli } t = 0, \\
    \alpha\,\beta^{t}, & \text{jeśli } t \ge 1.
\end{cases}
\]
\end{frame}

\begin{frame}{Polityki i funkcje wartości}
\textbf{Typy polityk:}
\begin{itemize}
    \item $\Phi$ -- polityki zależne od historii: $\phi_n : \mathcal{H}_n \rightarrow \Pr(A)$
    \item $\Phi_m$ -- polityki markowskie: $\phi_n : S \rightarrow \Pr(A)$
    \item $\Phi_s$ -- polityki stacjonarne: $\phi_s : S \rightarrow \Pr(A)$, $\phi_n = \phi_s$ dla każdego $n$
\end{itemize}

\vspace{0.5em}
\textbf{Funkcja wartości z dyskontowaniem QH:}
\begin{align*}
    J_{\phi}^{\alpha,\beta}(s_0)
    &= \mathbb{E}\!\left[ r(s_0,a_0) + \alpha\,\beta\, J_{\overrightarrow{\phi}}^{\beta}(s_1) \right] \\
    &= \sum_a \phi_0(a \mid s_0) \left[ r(s_0,a) + \alpha\,\beta\, \sum_{s'} q(s' \mid s_0,a) J_{\overrightarrow{\phi}}^{\beta}(s') \right]
\end{align*}
gdzie $\overrightarrow{\phi} = (\phi_1, \phi_2, \ldots)$, a $J_{\overrightarrow{\phi}}^{\beta}$ to wartość z dyskontowaniem wykładniczym.
\end{frame}

\begin{frame}{Przykład: Decyzja zakupowa}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Scenariusze:}
\begin{itemize}
    \item \textbf{Blisko:} Tanią rzecz teraz vs. lepszy produkt jutro
    \item \textbf{Daleko:} Tanią rzecz za 50 dni vs. lepszy produkt za 51 dni
\end{itemize}

\vspace{0.5em}
\textbf{Parametry:}
\begin{itemize}
    \item QH: $\alpha = 0.5$, $\beta = 0.99$
    \item EXP: $\alpha = 1.0$, $\beta = 0.99$
\end{itemize}

\vspace{0.5em}
\textbf{Wyniki:}
\begin{itemize}
    \item QH (blisko): kupuje tanią rzecz teraz
    \item QH (daleko): czeka na lepszy produkt
    \item EXP: zawsze czeka na lepszy produkt
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\centering
\includegraphics[width=0.95\textwidth]{../plots/choice_apple.png}
\end{column}
\end{columns}

\vspace{0.3em}
\textbf{Wniosek:} Dyskontowanie QH wykazuje niespójność czasową -- preferencje zmieniają się w zależności od horyzontu czasowego.
\end{frame}

\begin{frame}{Typy decydentów w zależności od $\alpha$}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{itemize}
    \item $\alpha = 1$ -- \textbf{decydent standardowy} (czasowo spójny)
    \item $0 < \alpha < 1$ -- \textbf{decydent quasi-hiperboliczny} (czasowo niespójny)
    \begin{itemize}
        \item \textit{Naiwny} -- nie przewiduje własnych przyszłych preferencji
        \item \textit{Zobowiązany} (precommitted) -- trzyma się wcześniej ustalonego planu
        \item \textit{Wyrafinowany} (sophisticated) -- przewiduje i bierze pod uwagę przyszłe zmiany preferencji
    \end{itemize}
    \item $\alpha = 0$ -- decydent krótkowzroczny
    \item $\alpha > 1$ -- decydent długowzroczny
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\centering
\includegraphics[width=0.95\textwidth]{../plots/qh_discounting_comparison.png}
\small Porównanie funkcji dyskontowania dla różnych wartości $\alpha$ przy $\beta = 0.95$
\end{column}
\end{columns}

\vspace{0.5em}
\textbf{W pracy:} Koncentrujemy się na decydentach zobowiązanych (precommitted agents).
\end{frame}
