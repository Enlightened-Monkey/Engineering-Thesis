\section{Wnioski}

\begin{frame}{Wnioski}
  \begin{itemize}
    \item Quasi-hiperboliczne dyskontowanie poprawia stabilność polityki w krótkim horyzoncie decyzji.
    \item Zbalansowany kij utrzymuje się dłużej niż w wersji z dyskontowaniem wykładniczym.
    \item Wpływ parametrów $\sigma$ i $\beta = \sigma\gamma$ na eksplorację wymaga dalszych badań.
  \end{itemize}
\end{frame}

\begin{frame}{Kierunki dalszych prac}
  \begin{itemize}
    \item Integracja z metodami policy gradient dla ciągłych akcji.
    \item Analiza odporności na szum i błędy percepcji stanu.
    \item Badanie scenariuszy z wieloma agentami i konfliktami preferencji.
  \end{itemize}
\end{frame}
