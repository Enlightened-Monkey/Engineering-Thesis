\section{Wstęp}

\begin{frame}{Motywacja}
  \begin{itemize}
    \item Decyzje w czasie wymagają modelowania preferencji krótkoterminowych i długoterminowych.
    \item Klasyczne dyskontowanie wykładnicze nie oddaje zjawiska prokrastynacji i zmian planów.
    \item Quasi-hiperboliczne dyskontowanie pozwala uchwycić silne uprzedzenia teraźniejszości.
  \end{itemize}
\end{frame}

\begin{frame}{Cele pracy}
  \begin{enumerate}
    \item Zbudować środowisko balansowania kija z losowo zmieniającymi się parametrami.
    \item Porównać standardowy Q-Learning z wersją quasi-hiperboliczną.
    \item Zinterpretować różnice jakości polityki i stabilności uczenia.
  \end{enumerate}
\end{frame}
